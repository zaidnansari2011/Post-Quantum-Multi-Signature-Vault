{# Flashed messages. A confirmation (success, info) is a toast: it is in the page, announced politely, and fades on its own (ui/feedback.html). A warning or an error stays inline until it is dismissed, because a message about something that went wrong must not disappear. #} {% from "ui/feedback.html" import toast %} {% with messages = get_flashed_messages(with_categories=true) %} {% set toasts = messages | selectattr(0, 'in', ['success', 'info', 'message']) | list %} {% for category, message in messages if category not in ('success', 'info', 'message') %}
{{ message }}
{% endfor %} {% if toasts %}
{% for category, message in toasts %}{{ toast(message, 'success' if category == 'success' else 'info') }}{% endfor %}
{% endif %} {% endwith %}
